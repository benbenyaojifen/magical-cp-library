    13void move_blood(vector<vector<char>>& v){
    14    //left 
    15    bool stuck = true;
    16    while(true){
    17    stuck = true;
    18    for(int i = 0; i < v.size(); i++){
    19        for(int j = 0; j < v[i].size(); j++){
    20            if(v[i][j] == 'W'){
    21                while(j > 0 && v[i][j - 1] == '.'){
    22                   v[i][j - 1] = 'W';
    23                   v[i][j] = '.';
    24                   j--;
    25                   stuck = false;
    26                }
    27            }
    28        }
    29    }
    30    //down 
    31        vector<pair<int,int>> drop;
    32        for (int i = 0; i < v.size() - 1; i++)
    33            for (int j = 0; j < v[i].size(); j++)
    34                if (v[i][j] == 'W' && v[i + 1][j] == '.')
    35                    drop.push_back({i, j});
    36 
    37        if (!drop.empty()) stuck = false;
    38        for (auto [i, j] : drop){
    39            v[i + 1][j] = 'W';
    40            v[i][j]   = '.';
    41        }
    42 
    43    if(stuck) break;
    44}
    45}
