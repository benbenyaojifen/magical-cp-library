The key move

Graph traversal

Explore states in breadth-first order with a queue, marking each state when it is reached so every vertex and edge is processed only as needed.

    21        q.pop();

    26            if (depth[v] == -1) {
    27                depth[v] = d + 1;

    31            }
    32        }
    33    }
    34    int ans = 0;
    35    for (int i = 0; i <= mx_depth; i++) {
    36        ans = max(ans, cnt_depth[i]);
    37    }
    38    for (int i = 1; i <= n; i++) {
    39        ans = max(ans, 1 + child_cnt[i]);
    40    }
