The key move

Shortest paths

Build a weighted graph, repeatedly expand the closest known state with a priority queue, and relax outgoing edges whenever a shorter route is found.

    7    ios::sync_with_stdio(0); cin.tie(0);

    10    for (int i = 0; i < m; i++) {
    11        int u, v, w; cin >> u >> v >> w;

    13        adj[v].emplace_back(w, u);
    14    }
    15    unordered_map<int, int> store;
    16    int k; cin >> k;
    17    for (int  i = 0; i < k; i++) {
    18        int v, w; cin >> v >> w; 
    19        store[v] = w;
    20    }
    21    int d; cin >> d;
    22    vector<int> dist(n + 1, inf);
    23    dist[d] = 0;
    24    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;
    25    pq.push({0, d});
    26    while (!pq.empty()) {
    27        auto [w, u] = pq.top(); pq.pop();
    28        if (w > dist[u]) continue;
    29        for (auto[wt, v] : adj[u]) {
    30            if (dist[v] > w + wt) {
    31                dist[v] = w + wt;
    32                pq.push({dist[v], v});
    33            }
    34        }
    35    }
    36    int ans = inf;
    37    for (int i = 1; i <= n; i++) {
    38        if (store.count(i)) ans = min(ans, dist[i] + store[i]);
    39    }
    40    cout << ans << '\n';
    41    return 0;
