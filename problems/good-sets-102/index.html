    3using namespace std;
    4const long long MOD = 1e9 + 7;
    5 
    6int main() {
    7    ios::sync_with_stdio(0);
    8    cin.tie(0);
    9    long long n, k;
    10    cin >> n >> k;
    11    vector<long long> dp(n + 1);
    12    dp[0] = 1; // empty set
    13    for (long long i = 1; i <= n; i++) {
    14        long long skip = dp[i - 1];
    15        long long take = (i >= k ? dp[i - k] : 1);
    16        dp[i] = (skip + take) % MOD;
    17    }
    18    cout << dp[n] % MOD << '\n';
    19    return 0;
    20}
