    14    int minR = 0, maxR = 0, maxC = 0;

    18    };

    24            r++;            
    25            add(r, c, '\\');

    30        }

    33    }
    34    for (int row = minR; row <= maxR; row++){
    35        auto &vec = rows[row];
    36        if (vec.empty()) continue;
    37        string line;
    38        int cur = 0;
    39        for (auto [col, ch] : vec){
    40            line.append(col - cur, '.');
    41            line.push_back(ch);
    42            cur = col + 1;
    43        }
    44        line.append(n - line.size(), '.');
