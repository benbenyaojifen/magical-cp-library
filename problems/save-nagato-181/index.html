The key move

Graph traversal

Explore states in breadth-first order with a queue, marking each state when it is reached so every vertex and edge is processed only as needed.

    4pair<int, vector<int>> bfs(int start, const vector<vector<int>> &g){
    5    int n = g.size() - 1;
    6    vector<int> d(n + 1, -1);
    7    queue<int> q;
    8    d[start] = 0; q.push(start);
    9    int far = start;
    10    while(!q.empty()){
    11        int u = q.front(); q.pop();
    12        if(d[u] > d[far]) far = u;
    13        for(int v : g[u]){
    14            if(d[v] == -1){
    15                d[v] = d[u] + 1;
    16                q.push(v);
    17            }
    18        }
    19    }
    20    return {far, d};
    21}
