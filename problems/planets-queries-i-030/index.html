    3int main(){

    5    int n, q; cin >> n >> q;
    6    const int MM = 30; // cuz 2 ^ 30 > k(1e9)
    7    vector<vector<int>> jump(MM, vector<int>(n + 1));
    8    for(int i = 1; i <= n; i++){
    9        int t; cin >> t;

    11    }
    12    for(int i = 1; i < MM; i++){
    13        for(int j = 1; j <= n; j++){
    14            jump[i][j] = jump[i - 1][jump[i - 1][j]];
    15        }
    16    }
    17    while(q--){
    18        int x; long long y; cin >> x >> y;
    19        for(int i = 0; i < MM; i++){
    20            if((y >> i) & 1LL) x = jump[i][x];
    21        }
    22        cout << x << '\n';
    23    }
    24    return 0;
