    3struct DSU {
    4    vector<int> parent;   
    5    vector<int> rank;      
    6 
    7    DSU(int n) : parent(n + 1), rank(n + 1, 0) {
    8        iota(parent.begin(), parent.end(), 0);  // parent[i] = i
    9    }
    10 
    11    int find(int x) {                      
    12        return parent[x] == x ? x : parent[x] = find(parent[x]);
    13    }
    14 
    15    void unite(int x, int y) {              
    16        int rootX = find(x);
    17        int rootY = find(y);
    18        if (rootX == rootY) return;      
    19 
    20        if (rank[rootX] > rank[rootY]) {    
    21            parent[rootY] = rootX;
    22        } else if (rank[rootX] < rank[rootY]) {
    23            parent[rootX] = rootY;
    24        } else {                            
    25            parent[rootY] = rootX;
    26            rank[rootX]++;                 
    27        }
    28    }
    29};
