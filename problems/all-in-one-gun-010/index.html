    20    }
    21    ll pref = 0;

    25        pref += v[m - 1];

    27        ll max_possible_damage = pref - min_p + max(min_p, max_s[m]);
    28        if (max_possible_damage >= r) {
    29            best_m = m;
    30            break;
    31        }
    32    }
    33    ll ans = cycles * (n + k) + best_m;
    34    cout << ans << '\n';
    35}
