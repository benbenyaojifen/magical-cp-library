    31void update(int idx, int l, int r, int pos, ll val) {
    32    if (l == r) {
    33        seg[idx].xr = val;
    34        return;

    36    int mid = (l + r) >> 1;
    37    if (pos <= mid) update(idx << 1, l, mid, pos, val);
    38    else update(idx << 1 | 1, mid + 1, r, pos, val);
    39    seg[idx] = mergeNode(seg[idx << 1], seg[idx << 1 | 1]);

    42int queryAbove(int pos) {
    43    int idx = 1, l = 0, r = N - 1;
    44    int res = 0;
    45 
    46    while (l < r) {

    49            int leftX = seg[idx << 1].xr;
    50            int rightX = seg[idx << 1 | 1].xr;

    52                res += (r - mid);
    53            }
