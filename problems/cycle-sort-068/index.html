    8vector<int> solve1(vector<int> a) {
    9    a.insert(a.end(), a.begin(), a.end());
    10    int idx1 = find(a.begin(), a.end(), 1) - a.begin();
    11    vector<int> b(a.begin() + idx1, a.end());
    12    for (int i = 0; i < n; i++) {
    13        if (b[i] != i + 1) {
    14            int miss = find(b.begin(), b.end(), i + 1) - b.begin();
    15            swap(b[i], b[miss]);

    17        }
    18    }

    24vector<int> solve2(vector<int> a) {
    25    int pos2 = find(a.begin(), a.end(), 2) - a.begin();
    26    int pos1 = find(a.begin(), a.end(), 1) - a.begin();
    27    // swap 1 into the slot just before 2
    28    swap(a[(pos2 - 1 + n) % n], a[pos1]);
    29    // rotate so that 1 is at front
    30    int start = find(a.begin(), a.end(), 1) - a.begin();
    31    vector<int> b;
    32    b.reserve(n);
    33    for (int i = 0; i < n; i++)
    34        b.push_back(a[(start + i) % n]);
    35    return b;
    36}
    37// if 1 is already right after 2, just swap them
    38vector<int> solve3(vector<int> a) {
    39    int pos2 = find(a.begin(), a.end(), 2) - a.begin();
    40    vector<int> b(n);

    44        swap(b[0], b[1]);
