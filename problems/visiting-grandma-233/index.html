    27    dist[1][c.count(1) ? 1 : 0] = 0;

    29    while(!pq.empty()) {

    31        if (w > dist[u][s]) continue;
    32        for (auto[wt, v] : adj[u]) {
    33            int ns = s || (c.count(v) ? 1 : 0);
    34            if (dist[v][ns] > w + wt) {
    35                dist[v][ns] = w + wt;
    36                cnt[v][ns] = cnt[u][s];
    37                pq.push({dist[v][ns], v, ns});
    38            } else if (dist[v][ns] == w + wt) {
    39                cnt[v][ns] = (cnt[v][ns] + cnt[u][s]) % mod;
    40            }
    41        }
    42    }
