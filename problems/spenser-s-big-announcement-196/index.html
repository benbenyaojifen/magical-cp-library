    18    auto check = [&] (ll w) {

    20        for (int i = 0; i < l && indx < v.size(); i++) {
    21            ll cur = w;
    22            while (indx < v.size() && cur >= v[indx]) {
    23                if (cur == w) {
    24                    cur -= v[indx];
    25                    indx++;
    26                } else {
    27                    if (cur >= v[indx] + s) {
    28                        cur -= v[indx] + s;
    29                        indx++;
    30                    } else {
    31                        break;
    32                    }
    33                }
    34            }
    35        }
    36        return indx == v.size();
    37    };
    38    ll ans = INF;
    39    while (lo <= hi) {
    40        ll mid = lo + (hi - lo) / 2;
    41        int res = check(mid);
    42        if (res) {
    43            ans = min(ans, mid);
    44            hi = mid - 1;
    45        } else {
    46            lo = mid + 1;
    47        }
    48    }
