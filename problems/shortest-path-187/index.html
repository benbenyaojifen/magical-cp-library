    5vector<int> bellman_ford(int v, vector<vector<int>>& edge, int src){
    6    vector<int> dist(v + 1, 1e8);
    7    dist[src] = 0;
    8    for(int i = 0; i < v; i++){
    9        for(auto e : edge){
    10            int u = e[0], v_ = e[1], w = e[2];
    11            if(dist[u] != 1e8 && dist[u] + w < dist[v_]){
    12                if(i == v - 1) return {-1};
    13                dist[v_] = dist[u] + w;
    14            }
    15        }
    16    }
    17    return dist;
    18}
