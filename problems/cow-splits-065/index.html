    21        string a = s.substr(l, 3), b = s.substr(r, 3);
    22        if (a == b) {
    23            for (int j = 0; j < 3; j++) {
    24                ans[l + j] = 1;
    25                ans[r + j] = 1;
    26            }
    27        } else {
    28            bool done = false;

    30                for (int k = j + 1; k < 3 && !done; k++) {
    31                    string t = "";
    32                    t += a[j]; t += a[k];
    33                    int p = 0;
    34                    for (char c : b) {
    35                        if (p < 2 && c == t[p]) p++;
    36                    }
    37                    if (p == 2) {
    38                        ans[l + j] = ans[l + k] = 1;
    39                        for (int x = 0; x < 3; x++) {
    40                            if (b[x] == t[0] || b[x] == t[1]) {
    41                                ans[r + x] = 1;
    42                            }
    43                        }
    44                        done = 1;
    45                    }
    46                }
    47            }
    48        }
    49    }
    50    cout << 2 << '\n';
    51    for (int i = 0; i < ans.size(); i++) {
    52        cout << ans[i] <<  " \n"[i == ans.size() - 1];
    53    }
    54}
