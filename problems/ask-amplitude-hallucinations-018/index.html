    9Node better(Node a, Node b) {
    10    if (a.val != b.val) return (a.val > b.val ? a : b);
    11    return (a.idx < b.idx ? a : b); // leftmost on tie
    12}

    14struct SegTree {
    15    int n;
    16    vector<Node> t;
    17 
    18    SegTree(const string &s) {
    19        n = s.size();
    20        t.resize(4*n);
    21        build(s, 1, 0, n-1);
    22    }
    23 
    24    void build(const string &s, int v, int l, int r) {
    25        if (l == r) {
    26            t[v] = {s[l] - '0', l};
    27            return;
    28        }
    29        int m = (l+r)/2;
    30        build(s, v*2, l, m);
    31        build(s, v*2+1, m+1, r);
    32        t[v] = better(t[v*2], t[v*2+1]);
    33    }
    34 
    35    Node query(int v, int l, int r, int ql, int qr) {
    36        if (qr < l || r < ql) return {-1, INT_MAX};
    37        if (ql <= l && r <= qr) return t[v];
    38        int m = (l+r)/2;
    39        return better(
    40            query(v*2, l, m, ql, qr),
    41            query(v*2+1, m+1, r, ql, qr)
    42        );
    43    }
    44 
    45    Node query(int l, int r) {
    46        return query(1, 0, n-1, l, r);
    47    }
    48};
