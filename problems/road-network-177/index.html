    11struct edge {
    12    int u, v; double d;
    13    edge(int u, int v, double d) : u(u), v(v), d(d){}
    14    bool operator<(const edge& other) const {

    18int find(int x) {
    19    if(parent[x] == x) return x;
    20    return parent[x] = find(parent[x]);
    21}
    22void unite(int x, int y) {
    23    x = find(x), y = find(y);
    24    parent[x] = y;
    25}

    32    vector<edge> e;

    44            auto[cc, xx, yy] = vv[j];

    58        }

    60    }
